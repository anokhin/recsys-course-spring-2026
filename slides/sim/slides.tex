\begin{frame}[shrink=6]{Цель симулятора}

\begin{columns}[T]
\begin{column}{0.57\textwidth}
\begin{tcolorbox}[colback=info!5,colframe=info!80,title=Зачем нужен симулятор]
\begin{itemize}
\item Приблизить поведение \textbf{реального пользователя}, а не только offline-метрики на готовом логе
\item Замкнуть цикл \textit{рекомендация $\rightarrow$ реакция $\rightarrow$ следующая рекомендация}
\item Проверять эффект обновлений рекомендательного сервиса
\end{itemize}
\end{tcolorbox}
\end{column}

\begin{column}{0.38\textwidth}
\begin{center}
\begin{tcolorbox}[colback=white,colframe=info!80,title=Главная идея]
\Large
\[
\hat r_t \Rightarrow
\text{feedback}_t \Rightarrow
\hat r_{t+1}
\]
\normalsize
\vspace{0.1cm}
Оценивать нужно не один айтем, а последовательность решений в сессии.
\end{tcolorbox}
\end{center}
\end{column}
\end{columns}

\vspace{0.15cm}

\begin{itemize}
\item Симулятор реализован как среда \textbf{Gymnasium}
\item Запуск разбит на условные \textbf{дни}: после каждого дня сервис можно переобучить и перезапустить
\end{itemize}

\begin{frame}{Алгоритм работы симулятора I: интересы пользователя}

\begin{columns}[T]
\begin{column}{0.5\textwidth}
\begin{tcolorbox}[colback=info!5,colframe=info!80,title=Профиль пользователя]
\footnotesize
\[
u = \{\code{interests}, b, s, B, \gamma\}
\]
\begin{itemize}
\item \code{interests} --- список треков-якорей
\item $b$ --- \code{consume\_bias}
\item $s$ --- \code{consume\_sharpness}
\item $B$ --- бюджет длины сессии
\item $\gamma$ --- штраф за повторы одного артиста
\end{itemize}
\end{tcolorbox}
\end{column}

\begin{column}{0.46\textwidth}
\begin{tcolorbox}[colback=info!5,colframe=info!80,title=Как стартует сессия]
\footnotesize
\[
t^\star \sim \mathrm{Uniform}(\code{interests}_u)
\]
\[
q_{\text{session}} = e(t^\star)
\]

\begin{itemize}
\item Из интересов случайно выбирается один трек-якорь
\item Его эмбеддинг становится \textbf{текущим намерением} сессии
\item Первый трек сессии сэмплируется из ближайших соседей этого эмбеддинга
\end{itemize}
\end{tcolorbox}
\end{column}
\end{columns}

\end{frame}

\begin{frame}{Алгоритм работы симулятора II: пользовательская сессия}

\begin{columns}[T]
\begin{column}{0.52\textwidth}
\footnotesize
\begin{enumerate}
\item Botify получает наблюдение \code{\{user, track\}} и возвращает следующий трек
\item Если трек уже встречался в сессии, время прослушивания равно $0$
\item Иначе считается близость рекомендации к эмбеддингу текущего интереса
\item Затем применяется штраф за повторения одного артиста
\item Награда = доля прослушивания; сессия завершается при исчерпании бюджета
\end{enumerate}
\end{column}

\begin{column}{0.44\textwidth}
\begin{tcolorbox}[colback=info!5,colframe=info!80,title=Формула реакции]
\footnotesize
\[
\text{score} = \langle e(i), q_{\text{session}} \rangle
\]
\[
\text{raw\_time} = \sigma((\text{score} - b) \cdot s)
\]
\[
\text{time} = \mathrm{round}\bigl(\text{raw\_time} \cdot \gamma^{n_{\text{artist}}}, 2\bigr)
\]

\begin{itemize}
\item $b$ управляет порогом ``нравится / не нравится''
\item $s$ задает резкость отклика
\item $\gamma^{n_{\text{artist}}}$ штрафует повторения одного артиста
\end{itemize}
\end{tcolorbox}
\end{column}
\end{columns}

\end{frame}

\begin{frame}{Как готовились данные симулятора}

\begin{tcolorbox}[colback=info!5,colframe=info!80,title=Генерация задачи с помощью LM]
\footnotesize
\begin{enumerate}
\item \textbf{\code{artists.py}}: собирает список популярных артистов по годам, жанрам и странам и сохраняет их метаданные.
\item \textbf{\code{tracks.py}}: для каждого артиста получает треки, очищает и нормализует данные и формирует каталог \code{tracks.json}.
\item \textbf{\code{embeddings.py}}: по описанию, жанрам, стране и summary трека строит его эмбеддинг и сохраняет общую матрицу векторов.
\item \textbf{\code{users.py}}: на основе каталога треков генерирует синтетических пользователей с интересами и параметрами отклика.
\end{enumerate}
\end{tcolorbox}

\vspace{0.15cm}
\footnotesize
Этот процесс идет от объектов предметной области к представлениям и только затем к пользователям: сначала строится музыкальный каталог, потом его векторное пространство, и уже в нем задаются пользовательские интересы.

\end{frame}
